# Mount backends: libfuse3 on Linux when present (dynamic link, LGPL; see DECISIONS D9),
# WinFsp's fuse3 layer on Windows, and the built-in NFSv3 loopback server on macOS
# (mount_nfs, no kext). The NFS server itself is portable and always built.
set(STEIN_MOUNT_SOURCES src/mount_none.cpp src/nfs_server.cpp)
set(STEIN_MOUNT_LIBS)
set(STEIN_HAVE_FUSE OFF)
set(STEIN_HAVE_NFS_MOUNT OFF)
if(APPLE)
  set(STEIN_MOUNT_SOURCES src/mount_nfs.cpp src/nfs_server.cpp)
  set(STEIN_HAVE_NFS_MOUNT ON)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(FUSE3 QUIET IMPORTED_TARGET fuse3)
    if(FUSE3_FOUND)
      set(STEIN_MOUNT_SOURCES src/mount_fuse.cpp src/nfs_server.cpp)
      set(STEIN_MOUNT_LIBS PkgConfig::FUSE3)
      set(STEIN_HAVE_FUSE ON)
    endif()
  endif()
elseif(WIN32)
  # WinFsp (GPLv3 with FLOSS exception; MIT consumers qualify) exposes a libfuse3-compatible
  # API. Its DLL is delay-loaded so the binaries run without WinFsp installed.
  set(_winfsp_root "$ENV{ProgramFiles\(x86\)}/WinFsp")
  if(NOT EXISTS "${_winfsp_root}/inc/fuse3/fuse.h")
    set(_winfsp_root "C:/Program Files (x86)/WinFsp")
  endif()
  if(EXISTS "${_winfsp_root}/inc/fuse3/fuse.h")
    set(_winfsp_lib "${_winfsp_root}/lib/winfsp-x64.lib")
    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
      set(_winfsp_lib "${_winfsp_root}/lib/winfsp-x86.lib")
    endif()
    if(EXISTS "${_winfsp_lib}")
      set(STEIN_MOUNT_SOURCES src/mount_fuse.cpp src/nfs_server.cpp)
      set(STEIN_MOUNT_LIBS "${_winfsp_lib}" delayimp)
      set(STEIN_WINFSP_INC "${_winfsp_root}/inc")
      set(STEIN_HAVE_FUSE ON)
    endif()
  endif()
endif()
if(WIN32)
  list(APPEND STEIN_MOUNT_LIBS ws2_32)
endif()
message(STATUS "stein_mount: FUSE backend ${STEIN_HAVE_FUSE}, NFS mount backend ${STEIN_HAVE_NFS_MOUNT}")

stein_add_module(mount
  SOURCES ${STEIN_MOUNT_SOURCES}
  PUBLIC_DEPS stein::core stein::fs)
if(STEIN_MOUNT_LIBS)
  target_link_libraries(stein_mount PRIVATE ${STEIN_MOUNT_LIBS})
endif()
if(STEIN_HAVE_FUSE)
  target_compile_definitions(stein_mount PUBLIC STEIN_HAVE_FUSE=1)
endif()
if(STEIN_HAVE_NFS_MOUNT)
  target_compile_definitions(stein_mount PUBLIC STEIN_HAVE_NFS_MOUNT=1)
endif()
if(STEIN_WINFSP_INC)
  target_include_directories(stein_mount PRIVATE "${STEIN_WINFSP_INC}")
  target_link_options(stein_mount PRIVATE "/DELAYLOAD:winfsp-x64.dll")
endif()

stein_add_test(test_mount
  SOURCES ../core/tests/test_main.cpp tests/test_mount.cpp
  DEPS stein::mount stein::fs)
if(TARGET test_mount)
  # A FUSE request that is never answered blocks uninterruptibly; fail instead of hanging CI.
  set_tests_properties(test_mount PROPERTIES TIMEOUT 180)
endif()
