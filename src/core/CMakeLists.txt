# The hardware kernels are compiled with their ISA extension enabled; they run
# only after cpuFeatures() confirms the CPU has it.
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64|i686")
    set_source_files_properties(src/crc32_hw.cpp PROPERTIES COMPILE_OPTIONS "-msse4.2")
    set_source_files_properties(src/sha256_hw.cpp PROPERTIES COMPILE_OPTIONS "-mssse3;-msse4.1;-msha")
    set_source_files_properties(src/aes_hw.cpp PROPERTIES COMPILE_OPTIONS "-msse2;-maes")
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
    set_source_files_properties(src/crc32_hw.cpp PROPERTIES COMPILE_OPTIONS "-march=armv8-a+crc")
    set_source_files_properties(src/sha256_hw.cpp PROPERTIES COMPILE_OPTIONS "-march=armv8-a+crypto")
    set_source_files_properties(src/aes_hw.cpp PROPERTIES COMPILE_OPTIONS "-march=armv8-a+crypto")
  endif()
endif()

if(WIN32)
  target_link_libraries(stein_core PRIVATE bcrypt)
endif()

# Expected<T> (include/stein/core/error.hpp) is std::expected where the standard
# library has it and the bundled tl::expected otherwise. The decision is made
# here and written into the generated expected_config.hpp, which is installed
# with the other headers, so consumers use the type the library was built with.
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("
#include <version>
#if !defined(__cpp_lib_expected) || __cpp_lib_expected < 202202L
#error std::expected is not available
#endif
#include <expected>
int main() { std::expected<int, int> e{1}; return *e - 1; }
" STEIN_HAVE_STD_EXPECTED)
if(STEIN_FORCE_TL_EXPECTED)
  set(STEIN_BUILT_WITH_TL_EXPECTED 1)
  message(STATUS "stein: STEIN_FORCE_TL_EXPECTED is ON, Expected<T> is tl::expected")
elseif(STEIN_HAVE_STD_EXPECTED)
  set(STEIN_BUILT_WITH_TL_EXPECTED 0)
else()
  set(STEIN_BUILT_WITH_TL_EXPECTED 1)
  message(WARNING
    "stein: std::expected is not available with ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION} "
    "and its standard library. Clang before 19 on GCC's libstdc++ is the usual case: libstdc++ hides "
    "<expected> from it. The build continues with the bundled tl::expected (third_party/expected). "
    "Please update to Clang 19 or newer, or build with GCC 13 or newer.")
endif()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/include/stein/core/expected_config.hpp CONTENT
"// Generated by src/core/CMakeLists.txt: which Expected<T> implementation this libstein was built with.
#pragma once
#define STEIN_BUILT_WITH_TL_EXPECTED @STEIN_BUILT_WITH_TL_EXPECTED@
" @ONLY)
target_include_directories(stein_core SYSTEM PUBLIC
  $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/third_party/expected>)
install(FILES ${PROJECT_SOURCE_DIR}/third_party/expected/tl/expected.hpp
  DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/tl)
